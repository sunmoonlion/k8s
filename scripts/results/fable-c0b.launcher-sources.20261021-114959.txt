function Read-C0Text([string]$Path) {
    $bytes = [IO.File]::ReadAllBytes($Path)
    if ($bytes.Length -eq 0) { return '' }
    if (($bytes.Length -ge 2 -and $bytes[0] -eq 255 -and $bytes[1] -eq 254) -or
        ($bytes.Length -ge 4 -and $bytes[1] -eq 0 -and $bytes[3] -eq 0)) {
        return [Text.Encoding]::Unicode.GetString($bytes).TrimStart([char]0xFEFF)
    }
    $utf8 = [Text.Encoding]::UTF8.GetString($bytes)
    if ($utf8.Contains([string][char]0xFFFD)) { return [Text.Encoding]::Default.GetString($bytes) }
    return $utf8.TrimStart([char]0xFEFF)
}

function Get-C0Clients {

function Assert-C0Window([string]$ApprovedAtUtc) {

function Invoke-C0Native($Context, [string]$Step, [string]$Exe, [string[]]$Arguments) {
    # Every native argument used by this fixed maintenance job is a single token.
    # Refuse ambiguous quoting rather than changing a target path silently.
    foreach ($arg in $Arguments) {
        if ($arg -match '[\s"]') { throw ('Unexpected native argument requiring quoting: ' + $Step) }
    }
    $Context.Index++
    $stem = '{0:D3}-{1}' -f $Context.Index,$Step
    $stdout = Join-Path $Context.RunDir ($stem + '.stdout.log')
    $stderr = Join-Path $Context.RunDir ($stem + '.stderr.log')
    Write-Host ('[' + (Get-Date -Format HH:mm:ss) + '] ' + $Step)
    $p = Start-Process -FilePath $Exe -ArgumentList ($Arguments -join ' ') -WorkingDirectory $Context.RunDir -NoNewWindow -PassThru -Wait -RedirectStandardOutput $stdout -RedirectStandardError $stderr
    $result = [pscustomobject]@{Code=$p.ExitCode; Out=(Read-C0Text $stdout); Err=(Read-C0Text $stderr); Stdout=$stdout; Stderr=$stderr}
    Add-Content -LiteralPath (Join-Path $Context.RunDir 'steps.log') -Value ($stem + ' exit=' + $result.Code)
    if ($result.Code -ne 0) { throw ($Step + ' failed, exit=' + $result.Code + '. See ' + $stdout + ' and ' + $stderr) }
    return $result
}

function Invoke-C0Wsl($Context, [string]$Step, [string[]]$Command) {

function Assert-C0NoRunningDistros($Context) {

function Save-C0Error([string]$Path, $Record) {
    # Append; never overwrite an earlier failure or depend on a later recovery.
    $text = [DateTime]::UtcNow.ToString('o') + "`r`n" + $Record.ToString() + "`r`n" + $Record.ScriptStackTrace + "`r`n"
    Add-Content -LiteralPath $Path -Value $text -Encoding UTF8
}

function Invoke-C0OfflineRecovery([scriptblock]$Offline, [scriptblock]$Recovery, [string]$RunDir) {
    $offlineFailure = $null
    $recoveryFailure = $null
    try { & $Offline } catch {
        $offlineFailure = $_.Exception.Message
        Save-C0Error -Path (Join-Path $RunDir 'offline-error.log') -Record $_
    }
    try { & $Recovery } catch {
        $recoveryFailure = $_.Exception.Message
        Save-C0Error -Path (Join-Path $RunDir 'recovery-error.log') -Record $_
    }
    [pscustomobject]@{OfflineError=$offlineFailure; RecoveryError=$recoveryFailure; Success=($null -eq $offlineFailure -and $null -eq $recoveryFailure)}
}

function Assert-C0DiskpartSuccess([int]$Code, [string]$Text) {
    if ($Code -ne 0 -or $Text -match '(?i)\berror\b|failed|失败|错误|拒绝|无法|不支持|找不到' -or
        $Text -notmatch '(?is)successfully.*compact|成功.*压缩') {
        throw 'DiskPart success was not established. Preserve its stdout/stderr; do not retry blindly.'
    }
}

→ DiskPart compact 原系统盘和 sunmoon-data.vhdx
→ 原附盘任务先附盘再启动 Ubuntu → preflight/platform-start
→ Jenkins/代理启动检查 → platform-status/节点检查。
不删除数据，不调整虚拟容量、UUID、服务身份或部署版本。

运行中保持脚本窗口打开，不启动 VS Code、Ubuntu、WSL 标签页或访问 WSL 目录。
压缩时不要中断 DiskPart。控制台会显示当前步骤和日志位置。
恢复检查可能需要几分钟；它要等待应用 Pod 和 Flux 全部就绪。
看到绿色 Completed、exit=0 后才重新打开 VS Code/WSL，告诉助手“回来了”。
若 exit=1，先保存日志；等脚本结束后再重新连接助手，不盲目再次压缩。

日志每轮单独保留：
C:\wsl-disks\maintenance\c0b-20261021\runs\<开始时间>\
last-run.txt 记录最近一轮目录。每条原生命令分别保存 stdout/stderr 和退出码。
before.json / after-compact.json / after.json 记录 C 空闲及两块 VHDX 长度；
result.json 和 result.log 记录结果。offline-error.log 在开始恢复前立即保存
最初离线异常；recovery-error.log 单独保存恢复失败，互不覆盖。

只恢复服务（不再次压缩），需要先检查失败日志；用管理员 Windows PowerShell：
powershell.exe -NoProfile -ExecutionPolicy Bypass -File C:\wsl-disks\maintenance\c0b-20261021\Start-C0b.ps1 -Recover C:\wsl-disks\maintenance\c0b-20261021\runs\<实际开始时间>
不要原样复制尖括号，必须用该轮真实目录。恢复同样走现有附盘任务和 Make 入口。

只读预览（可在准备阶段使用，不会停服务、trim 或压缩）：
powershell.exe -NoProfile -ExecutionPolicy Bypass -File C:\wsl-disks\maintenance\c0b-20261021\Start-C0b.ps1
